library(vegan)
library(ggplot2)
library(ggrepel)

# Load metadata
metadata <- read.csv("metadata.csv", header = TRUE, check.names = FALSE)
rownames(metadata) <- metadata$Samples
metadata$Samples <- NULL

# Load viral contig data
mag_raw <- read.csv("mag_abundance.csv", header = TRUE, row.names = 1, check.names = FALSE)

# Transpose so samples are rows
mag_t <- t(mag_raw)

# Align samples
common <- intersect(rownames(mag_t), rownames(metadata))
mag_t <- mag_t[common, , drop = FALSE]
metadata <- metadata[common, , drop = FALSE]

# Hellinger transform
spe.hel <- decostand(mag_t, method = "hellinger")

# -----------------------------
# Focused environmental variables
# -----------------------------
env <- metadata[, c("Total_C", "Total_N", "P", "NAG", "AP"), drop = FALSE]

# Ensure numeric
env <- as.data.frame(env)
env[] <- lapply(env, function(x) as.numeric(as.character(x)))

# Remove missing rows
keep <- complete.cases(env)
env2 <- env[keep, , drop = FALSE]
spe.hel <- spe.hel[keep, , drop = FALSE]
metadata <- metadata[keep, , drop = FALSE]

# Confirm alignment
stopifnot(all(rownames(spe.hel) == rownames(metadata)))

# Standardize
env.z <- decostand(env2, method = "stand")

# -----------------------------
# RDA (no forward selection needed)
# -----------------------------
spe.rda.signif <- rda(spe.hel ~ Total_C + Total_N + P + NAG + AP, data = env.z)

print(RsquareAdj(spe.rda.signif))
anova.cca(spe.rda.signif, permutations = 1000)
anova.cca(spe.rda.signif, permutations = 1000, by = "term")
anova.cca(spe.rda.signif, permutations = 1000, by = "axis")

# -----------------------------
# Triplot
# -----------------------------
site_scores <- scores(spe.rda.signif, display = "sites", scaling = 1)
sp_scores   <- scores(spe.rda.signif, display = "species", scaling = 1)
bp_scores   <- scores(spe.rda.signif, display = "bp", scaling = 1)

perc <- round(100 * summary(spe.rda.signif)$cont$importance[2, 1:2], 2)

plot(spe.rda.signif,
     scaling = 1, type = "none", frame = FALSE,
     xlab = paste0("RDA1 (", perc[1], "%)"),
     ylab = paste0("RDA2 (", perc[2], "%)"))

points(site_scores, pch = 21, col = "black", bg = "darkorange", cex = 1.3)
points(sp_scores, pch = 22, col = "black", bg = "blue", cex = 1.1)

high_score_mags <- rownames(sp_scores)[rowSums(abs(sp_scores) > 0.35) > 0]
if (length(high_score_mags) > 0) {
  text(sp_scores[high_score_mags, , drop = FALSE],
       labels = high_score_mags,
       col = "grey4", cex = 0.6, pos = 3)
}

arrow_scale <- 3

if (nrow(bp_scores) > 0) {
  arrows(0, 0,
         bp_scores[,1] * arrow_scale,
         bp_scores[,2] * arrow_scale,
         col = "red", lwd = 1.2)
  
  text(bp_scores[,1] * arrow_scale * 1.15,
       bp_scores[,2] * arrow_scale * 1.15,
       labels = rownames(bp_scores),
       col = "red", cex = 0.9, font = 2, pos = 4)
}

# -----------------------------
# dbRDA (Bray-Curtis)
# -----------------------------
bc_dist <- vegdist(mag_t, method = "bray")

spe.dbrda <- dbrda(bc_dist ~ Total_C + Total_N + P + NAG + AP, data = env.z)

print(RsquareAdj(spe.dbrda))
anova.cca(spe.dbrda, permutations = 1000)
anova.cca(spe.dbrda, permutations = 1000, by = "term")
anova.cca(spe.dbrda, permutations = 1000, by = "axis")

#----
add legend 
#----
legend("topright",
       legend = c("Samples", "Viral contigs", "Env variables"),
       pch = c(21, 22, NA),
       pt.bg = c("darkorange", "blue", NA),
       col = c("black", "black", "red"),
       lty = c(NA, NA, 1),
       pt.cex = 1.2,
       cex = 0.9,
       bty = "n")
#add sample names
text(site_scores,
     labels = rownames(site_scores),
     cex = 0.7,
     col = "black",
     pos = 3)

#add stats to the plot
R2 <- RsquareAdj(spe.dbrda)$adj.r.squared
pval <- anova.cca(spe.dbrda)$`Pr(>F)`[1]
mtext(paste0("dbRDA (Bray-Curtis) | Adj R² = ",
             round(R2, 3),
             " | p = ",
             signif(pval, 2)),
      side = 3, line = 1, cex = 0.9)
